def fast_ransac_circle_fit(points, iterations=500, threshold=10):
    """优化的RANSAC圆拟合"""
    n_points = len(points)
    if n_points < 3:
        return None
    best_inliers_count = 0
    best_params = None

    points_x = points[:, 0]
    points_y = points[:, 1]

    for _ in range(iterations):
        idx = np.random.choice(n_points, 3, replace=False)
        p1, p2, p3 = points[idx]

        x1, y1 = p1
        x2, y2 = p2
        x3, y3 = p3

        det = 2 * (x1 * (y2 - y3) - y1 * (x2 - x3) + x2 * y3 - x3 * y2)

        if abs(det) < 1e-10:
            continue

        k1 = x1**2 + y1**2
        k2 = x2**2 + y2**2
        k3 = x3**2 + y3**2

        cx = (k1 * (y2 - y3) + k2 * (y3 - y1) + k3 * (y1 - y2)) / det
        cy = (k1 * (x3 - x2) + k2 * (x1 - x3) + k3 * (x2 - x1)) / det

        r = np.sqrt((x1 - cx)**2 + (y1 - cy)**2)

        if r <= 0 or r > 2000:
            continue

        distances = np.abs(np.sqrt((points_x - cx)**2 + (points_y - cy)**2) - r)
        inliers_count = np.sum(distances < threshold)

        if inliers_count > best_inliers_count:
            best_inliers_count = inliers_count
            best_params = (cx, cy, r)
    return best_params


def detect_and_mask_circle(image):
    """
    检测圆形并创建掩码图像
    参数:
        image: numpy数组格式的图像(灰度或彩色)
    返回:
        masked_img: 掩码后的图像(圆外部分为黑色)
        circle_info: 圆的信息字典
    """
    if image is None:
        print(f"无法处理空图像")
        return None, None

    # 如果是彩色图像，转换为灰度
    if len(image.shape) == 3:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        img = image.copy()
    else:
        gray = image.copy()
        img = cv2.cvtColor(gray, cv2.COLOR_GRAY2BGR)

    blurred = cv2.GaussianBlur(gray, (5, 5), 2)

    h, w = gray.shape
    center_y, center_x = h // 2, w // 2

    angles = np.linspace(0, 2*np.pi, 180)
    cos_angles = np.cos(angles)
    sin_angles = np.sin(angles)

    max_radius = min(h, w) // 2
    radii = np.arange(5, max_radius, 2)

    threshold = np.mean(gray) + 0.5 * np.std(gray)

    outer_edge_points = []

    for i, (cos_a, sin_a) in enumerate(zip(cos_angles, sin_angles)):
        x_coords = center_x + radii * cos_a
        y_coords = center_y + radii * sin_a

        valid_mask = (x_coords >= 0) & (x_coords < w) & \
                     (y_coords >= 0) & (y_coords < h)

        if not np.any(valid_mask):
            continue

        x_coords = x_coords[valid_mask].astype(int)
        y_coords = y_coords[valid_mask].astype(int)

        intensities = blurred[y_coords, x_coords]
        bright_indices = np.where(intensities > threshold)[0]

        if len(bright_indices) > 0:
            farthest_idx = bright_indices[-1]
            outer_edge_points.append([x_coords[farthest_idx], y_coords[farthest_idx]])

    if len(outer_edge_points) < 10:
        mean_val = np.mean(blurred)
        std_val = np.std(blurred)
        threshold = mean_val + 0.3 * std_val
        _, bright_mask = cv2.threshold(blurred, threshold, 255, cv2.THRESH_BINARY)

        contours, _ = cv2.findContours(bright_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

        if contours:
            largest_contour = max(contours, key=cv2.contourArea)
            outer_edge_points = largest_contour[::5].reshape(-1, 2)
        else:
            return None, None

    outer_edge_points = np.array(outer_edge_points)
    best_circle = fast_ransac_circle_fit(outer_edge_points, iterations=500)

    if best_circle is None:
        return None, None

    xc, yc, C = best_circle
    R_extended = C + 20

    # 创建圆形掩码
    mask = np.zeros(img.shape[:2], dtype=np.uint8)
    cv2.circle(mask, (int(xc), int(yc)), int(R_extended), 255, -1)

    # 应用掩码
    masked_img = cv2.bitwise_and(img, img, mask=mask)

    circle_info = {
        'center': (xc, yc),
        'radius': C,
        'extended_radius': R_extended
    }

    return masked_img, circle_info

    def load_defect_model(self):

    def process_image_with_sahi(self, frame, conf_threshold=0.25, slice_height=1024, slice_width=1024):
        """使用SAHI进行切片检测处理，集成圆形拟合和不同颜色的锚框"""
        try:
            # ===== 步骤1: 圆形拟合和掩码处理 =====
            print("  >> 执行圆形拟合...")
            masked_frame, circle_info = detect_and_mask_circle(frame)

            if masked_frame is None or circle_info is None:
                print("  >> 警告: 圆形拟合失败，使用原始图像进行检测")
                masked_frame = frame
                circle_info = None
            else:
                center_x, center_y = circle_info['center']
                radius = circle_info['radius']
                print(f"  >> 圆形拟合成功 - 圆心: ({center_x:.1f}, {center_y:.1f}), 半径: {radius:.1f}px")

            # ===== 步骤2: 转换为RGB进行检测 =====
            if len(masked_frame.shape) == 2:
                # 灰度图转RGB
                rgb_frame = cv2.cvtColor(masked_frame, cv2.COLOR_GRAY2RGB)
            else:
                rgb_frame = masked_frame.copy()

            # ===== 步骤3: 使用SAHI进行切片检测 =====
            print("  >> 执行SAHI切片检测...")
            result = get_sliced_prediction(
                rgb_frame,
                self.model,
                slice_height=slice_height,
                slice_width=slice_width,
                overlap_height_ratio=0.1,
                overlap_width_ratio=0.1,
                perform_standard_pred=True,
                postprocess_type="NMS",
                postprocess_match_threshold=0.5,
                postprocess_match_metric="IOU"
            )

            # ===== 步骤4: 定义不同类型瑕疵的颜色 =====
            defect_colors = {
                'splash': (0, 165, 255),    # 黄色 - 麻点
                'scratch': (255, 0, 255),   # 洋红 - 划痕
                'chipping': (255, 165, 0),  # 橙色 - 崩边
                'default': (0, 0, 255)      # 默认红色
            }

            # ===== 步骤5: 创建标注器并绘制结果 =====
            annotator = Annotator(rgb_frame, line_width=2)

            detection_data = [
                (det.category.name, det.category.id, (det.bbox.minx, det.bbox.miny, det.bbox.maxx, det.bbox.maxy),
                 float(det.score.value))
                for det in result.object_prediction_list
            ]

            cnt = 0
            defect_details = []

            for det in detection_data:
                cnt += 1
                minx, miny, maxx, maxy = det[2]
                confidence = det[3]
                width = maxx - minx
                height = maxy - miny
                defect_type = det[0]

                defect_info = {
                    "id": cnt, "category": defect_type, "category_id": int(det[1]),
                    "confidence": float(confidence),
                    "bbox": {
                        "x1": float(minx), "y1": float(miny), "x2": float(maxx),
                        "y2": float(maxy), "width": float(width), "height": float(height)
                    }
                }
                defect_details.append(defect_info)

                # 根据瑕疵类型选择颜色
                color = defect_colors.get(defect_type, defect_colors['default'])

                label = f"{defect_type} {confidence:.2f}"
                annotator.box_label(det[2], label=label, color=color)

            result_img = annotator.result()

            # ===== 步骤6: 如果进行了圆形拟合，在结果图上绘制圆形边界 =====
            if circle_info is not None:
                center_x, center_y = int(circle_info['center'][0]), int(circle_info['center'][1])
                radius = int(circle_info['radius'])
                # 绘制圆形边界（白色，半透明）
                cv2.circle(result_img, (center_x, center_y), radius, (255, 255, 255), 2)
                # 绘制圆心
                cv2.circle(result_img, (center_x, center_y), 5, (0, 255, 0), -1)

            print(f"  >> 检测完成，共检测到 {cnt} 个瑕疵")
            return result_img, result, cnt, defect_details

        except Exception as e:
            print(f"处理图像时出错: {e}")
            import traceback
            traceback.print_exc()
            return None, None, 0, []

                                     data['DA3827093']['pic_path']]
